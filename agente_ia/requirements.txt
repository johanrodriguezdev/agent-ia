# --- Terminal embebida ---
# La pseudo-consola real del sistema: ConPTY en Windows, PTY en Linux/macOS. Con tuberias
# normales el shell detecta que no hay consola y apaga colores, autocompletado y edicion de
# linea; con una PTY la consola del panel es una terminal de verdad. Cada paquete se instala
# solo donde corresponde. Si falta, la app arranca igual y el panel avisa cual falta
# (core/terminal_session.py). El lado del navegador (xterm.js) va vendorizado en
# ui/webview/frontend/vendor/.
pywinpty; sys_platform == "win32"
ptyprocess; sys_platform != "win32"
# OCR con el motor de Windows (Windows.Media.Ocr): leer texto dibujado en pantalla,
# lienzos, imagenes y PDF escaneados sin modelo ni red. Solo Windows.
winsdk; sys_platform == "win32"
# Dibujar las paginas de un PDF escaneado para pasarlas por el OCR.
pymupdf

python-pptx
